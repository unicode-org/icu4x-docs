100extern "C" inline void _flush(capi::DiplomatWrite* w) {
101 std::string* string = reinterpret_cast<std::string*>(w->context);

105extern "C" inline bool _grow_impl(capi::DiplomatWrite* w, uintptr_t requested) {
106 std::string* string = reinterpret_cast<std::string*>(w->context);
107 string->resize(requested);
108 w->cap = string->length();

117inline capi::DiplomatWrite WriteFromString(std::string& string) {
118 capi::DiplomatWrite w;
119 w.context = &string;
120 w.buf = &string[0];
121 w.len = string.length();
122 w.cap = string.length();
123 // Will never become true, as _grow is infallible.
124 w.grow_failed = false;
125 w.flush = _flush;
126 w.grow = _grow;

462 // For a given T, creates a function that take in the C ABI version & return the C++ type.
463 template<typename T>
464 static T replace(replace_fn_t<T> val) {
465 if constexpr(std::is_same_v<T, std::string_view>) {
466 return std::string_view{val.data, val.len};
467 } else if constexpr (!std::is_same_v<T, diplomat_c_span_convert_t<T>>) {
468 return T{ val.data, val.len };
469 } else if constexpr (!std::is_same_v<T, as_ffi_t<T>>) {
470 if constexpr (std::is_lvalue_reference_v<T>) {
471 return *std::remove_reference_t<T>::FromFFI(val);
472 } else if constexpr (is_unique_ptr_v<T>) {
473 return T(is_unique_ptr<T>::type::FromFFI(val));
474 }
475 else {
476 return T::FromFFI(val);
477 }
478 }
479 else {
480 return val;
481 }
482 }
483
484 template<typename T>
485 static replace_fn_t<T> replace_ret(T val) {
486 if constexpr(std::is_same_v<T, std::string_view>) {
487 return {val.data(), val.size()};
488 } else if constexpr (!std::is_same_v<T, diplomat_c_span_convert_t<T>>) {
489 // Can we convert straight away to our slice type, or (in the case of ABI compatible structs), do we have to do a reinterpret cast?
490 if constexpr(std::is_same_v<decltype(std::declval<T>().data()), decltype(replace_fn_t<T>::data)>) {
491 return replace_fn_t<T> { val.data(), val.size() };
492 } else {
493 return replace_fn_t<T> { reinterpret_cast<decltype(replace_fn_t<T>::data)>(val.data()), val.size() };
494 }
495 } else if constexpr(!std::is_same_v<T, as_ffi_t<T>>) {
496 return val.AsFFI();
497 } else {
498 return val;
499 }

523 template<typename T, typename E, typename TOut>
524 static TOut c_run_callback_result(const void *cb, replace_fn_t<Args>... args) {
525 result<T, E> res = c_run_callback(cb, args...);
526
527 auto is_ok = res.is_ok();
528

539 } else {

542 }

549 } else {

551 }
552 }
553 }
